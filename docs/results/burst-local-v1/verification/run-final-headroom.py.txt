import json,pathlib,subprocess,sys,time
status=pathlib.Path('work/burst-results/final-status.json')
for repeat in range(1,4):
 out=pathlib.Path(f'work/burst-results/final-256-r{repeat}')
 info={'case':out.name,'vus':256,'profile':'burst','generatorMemoryMiB':1536,'repeat':repeat,'startedAtEpoch':time.time(),'stage':'running'}
 status.write_text(json.dumps(info,indent=2)+'\n');print('Starting '+out.name,flush=True)
 result=subprocess.run([sys.executable,'work/cohort-headroom.py',str(out),'256','burst','300'])
 info.update(stage='finished' if not result.returncode else 'failed',exitCode=result.returncode,finishedAtEpoch=time.time());status.write_text(json.dumps(info,indent=2)+'\n')
 if result.returncode:sys.exit(result.returncode)
print('All three new256VU headroom windows completed.',flush=True)
