import json,pathlib,subprocess,sys,time
status=pathlib.Path('work/burst-results/formal-status.json')
for vus,profile in [('64','standard'),('256','burst')]:
 for repeat in range(1,4):
  out=pathlib.Path(f'work/burst-results/formal-{vus}-r{repeat}')
  info={'case':out.name,'vus':int(vus),'profile':profile,'repeat':repeat,'startedAtEpoch':time.time(),'stage':'running'}
  status.write_text(json.dumps(info,indent=2)+'\n');print('Starting '+out.name,flush=True)
  result=subprocess.run([sys.executable,'work/run-burst-cohort.py',str(out),vus,profile,'300'])
  info.update(stage='finished' if not result.returncode else 'failed',exitCode=result.returncode,finishedAtEpoch=time.time())
  status.write_text(json.dumps(info,indent=2)+'\n')
  if result.returncode:sys.exit(result.returncode)
print('All six formal windows completed.',flush=True)
