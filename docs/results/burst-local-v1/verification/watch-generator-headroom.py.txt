#!/usr/bin/env python3
"""Read-only observation of the generator mounted to one exact results directory."""
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import subprocess
import sys
import time

CASE = Path(sys.argv[1]).resolve()
PREFIX = str(CASE)
STOP = Path(PREFIX + '.generator-observer.stop')
META = Path(PREFIX + '.generator-observer.json')
START = Path(PREFIX + '.generator-start.json')
SAMPLES = Path(PREFIX + '.generator-cgroup.jsonl')
FILES = ['/sys/fs/cgroup/memory.current', '/sys/fs/cgroup/memory.peak',
         '/sys/fs/cgroup/memory.max', '/sys/fs/cgroup/memory.events',
         '/sys/fs/cgroup/memory.pressure', '/sys/fs/cgroup/memory.swap.current',
         '/sys/fs/cgroup/memory.swap.max', '/sys/fs/cgroup/cpu.stat']
BEGAN = time.monotonic()


def stamp():
    return {'timeUtc': datetime.now(timezone.utc).isoformat(),
            'monotonicNs': time.monotonic_ns(),
            'observerElapsedSeconds': time.monotonic() - BEGAN}


def run(command):
    return subprocess.run(command, text=True, capture_output=True, timeout=2)


def save(metadata):
    META.write_text(json.dumps(metadata, indent=2) + '\n')


metadata = {**stamp(), 'scope': 'read-only generator state/resource observation',
            'casePath': str(CASE), 'selection': 'exact bind Source=casePath and Destination=/results',
            'observerSha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
            'targetIntervalSeconds': 1, 'execTimeoutSeconds': 2, 'cgroupFiles': FILES,
            'commandEnvironmentWorkloadChanges': False, 'state': 'discovering',
            'validated': False, 'sampleCount': 0, 'errorCount': 0,
            'limitations': ['Samples are not atomic across cgroup files.',
                            'Observer has Docker-exec and filesystem overhead.',
                            'No retrospective causal claim about the prior 768MiB failure.']}
save(metadata)
generator = None
script = 'for task_file in ' + ' '.join(FILES) + '; do printf "\\n@@ %s\\n" "$task_file"; if [ -r "$task_file" ]; then cat "$task_file"; else printf "<unavailable>\\n"; fi; done'

try:
    with SAMPLES.open('w', buffering=1) as stream:
        while not STOP.exists():
            began = time.monotonic()
            row = stamp()
            try:
                if generator is None:
                    environment = CASE / 'environment.json'
                    if not environment.exists():
                        time.sleep(0.25)
                        continue
                    env = json.loads(environment.read_text())
                    expected = {'imageId': env['k6Image'], 'memoryBytes': 1536 * 1024 * 1024,
                                'nanoCpus': 4_000_000_000}
                    if env['generator']['imageId'] != expected['imageId']:
                        raise ValueError('environment immutable generator identities disagree')
                    if env['generatorLimits']['memoryMiB'] != 1536 or env['generatorLimits']['cpus'] != 4:
                        raise ValueError('environment does not declare the intended 1536MiB/4CPU generator')
                    metadata['expected'] = expected
                    listed = run(['docker', 'ps', '-q'])
                    if listed.returncode:
                        raise RuntimeError(listed.stderr)
                    ids = listed.stdout.split()
                    matches = []
                    if ids:
                        inspected = run(['docker', 'inspect', *ids])
                        if inspected.returncode:
                            raise RuntimeError(inspected.stderr)
                        for item in json.loads(inspected.stdout):
                            if any(m.get('Type') == 'bind' and m.get('Destination') == '/results'
                                   and Path(m.get('Source', '')).resolve() == CASE for m in item.get('Mounts', [])):
                                matches.append(item)
                    if not matches:
                        time.sleep(0.5)
                        continue
                    if len(matches) != 1:
                        raise ValueError('multiple running containers have the exact results bind')
                    actual = matches[0]
                    subset = {**stamp(), 'Id': actual['Id'], 'Image': actual['Image'],
                              'Name': actual['Name'], 'State': actual['State'],
                              'HostConfig': {key: actual['HostConfig'].get(key) for key in
                                             ['Memory', 'MemorySwap', 'NanoCpus']},
                              'resultsBind': [m for m in actual['Mounts'] if m.get('Destination') == '/results'],
                              'Cmd': actual['Config'].get('Cmd'),
                              'GoRuntimeOverrides': [value for value in actual['Config'].get('Env', [])
                                  if value.startswith(('GODEBUG=', 'GOGC=', 'GOMEMLIMIT=', 'GOMAXPROCS='))]}
                    START.write_text(json.dumps(subset, indent=2) + '\n')
                    errors = []
                    if actual['Image'] != expected['imageId']:
                        errors.append('actual image differs from environment immutable image')
                    if subset['HostConfig']['Memory'] != expected['memoryBytes']:
                        errors.append('actual memory limit differs from 1536MiB')
                    if subset['HostConfig']['NanoCpus'] != expected['nanoCpus']:
                        errors.append('actual CPU limit differs from 4CPU')
                    if subset['Cmd'] != ['run', '--quiet', '/workspace/benchmarks/acceptance.js']:
                        errors.append('generator command differs from the unchanged runner command')
                    if subset['GoRuntimeOverrides']:
                        errors.append('unexpected Go runtime environment override')
                    metadata.update(generatorId=actual['Id'], generatorName=actual['Name'],
                                    validated=not errors, validationErrors=errors, state='observing')
                    save(metadata)
                    generator = actual['Id']

                sample = run(['docker', 'exec', generator, 'sh', '-c', script])
                row.update(returncode=sample.returncode, stdout=sample.stdout, stderr=sample.stderr)
                if sample.returncode:
                    row['error'] = 'docker exec failed; raw stderr retained, including EOF/removal'
                    metadata['errorCount'] += 1
            except Exception as failure:
                row['error'] = repr(failure)
                metadata['errorCount'] += 1
            row['durationSeconds'] = time.monotonic() - began
            stream.write(json.dumps(row) + '\n')
            metadata['sampleCount'] += 1
            time.sleep(max(0, 1 - (time.monotonic() - began)))
finally:
    metadata.update(finishedAtUtc=datetime.now(timezone.utc).isoformat(), state='stopped',
                    elapsedSeconds=time.monotonic() - BEGAN)
    save(metadata)
